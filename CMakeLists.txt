cmake_minimum_required(VERSION 3.30.4)


project(covoxslam VERSION 0.1.0 LANGUAGES CXX)


set(CMAKE_CXX_STANDARD 23)
 set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(covoxslam_core STATIC src/core/geometry.cpp src/input/json_reader.cpp src/input/dataset.cpp)
target_sources(covoxslam_core PRIVATE src/input/dataset_metadata.cpp)
target_sources(covoxslam_core PRIVATE src/input/toml_tables.cpp)
target_sources(covoxslam_core PRIVATE src/pipeline/frame_pipeline.cpp)
target_sources(covoxslam_core PRIVATE src/pipeline/mapping_session.cpp)
target_sources(covoxslam_core PRIVATE src/mapping/config.cpp)
target_sources(covoxslam_core PRIVATE src/voxel/tile_geometry.cpp)
target_sources(covoxslam_core PRIVATE src/voxel/sparse_map.cpp)
target_sources(covoxslam_core PRIVATE src/export.cpp)
target_sources(covoxslam_core PRIVATE src/sensor/lookup.cpp)
target_sources(covoxslam_core PRIVATE src/sensor/prepared_layout.cpp)
target_sources(covoxslam_core PRIVATE src/mapping/tsdf/integrator.cpp)
set(covoxslam_sensor_accumulation_bits "32" CACHE STRING "coVoxSLAM sensor accumulation precision")
set_property(CACHE covoxslam_sensor_accumulation_bits PROPERTY STRINGS 32)
target_compile_definitions(covoxslam_core PUBLIC
    covoxslam_sensor_accumulation_bits=${covoxslam_sensor_accumulation_bits})
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/covoxslam_sensor_precision.txt"
    CONTENT "gpu_accumulation_bits=${covoxslam_sensor_accumulation_bits}\nfield_storage_bits=32\n")
option(covoxslam_enable_cuda "Build coVoxSLAM GPU mapping and runtime support" OFF)
if(covoxslam_enable_cuda)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 75)
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit 12.6 REQUIRED)
    target_sources(covoxslam_core PRIVATE src/cuda_device.cpp cuda/foundation.cu)
    target_sources(covoxslam_core PRIVATE cuda/sensor_grid/prepared.cu)
    target_sources(covoxslam_core PRIVATE cuda/sensor_grid/transfer.cu)
    target_sources(covoxslam_core PRIVATE cuda/sensor_grid/discovery.cu)
    target_sources(covoxslam_core PRIVATE cuda/sensor_grid/resident.cu)
    target_sources(covoxslam_core PRIVATE cuda/storage/initialize.cu)
    target_sources(covoxslam_core PRIVATE cuda/store/index.cu)
    target_sources(covoxslam_core PRIVATE cuda/store/band_esdf.cu)
    target_sources(covoxslam_core PRIVATE src/storage/cuda_bridge.cpp)
    target_compile_definitions(covoxslam_core PUBLIC covoxslam_has_cuda=1)
    target_sources(covoxslam_core PRIVATE src/sensor/dense_field.cpp)
    target_sources(covoxslam_core PRIVATE src/mapping/tsdf/active_submap.cpp)
    target_sources(covoxslam_core PRIVATE src/sensor_grid_integrate.cpp)
    target_sources(covoxslam_core PRIVATE src/sensor_grid_resident.cpp)
    target_sources(covoxslam_core PRIVATE src/sensor_grid_soa.cpp)
    target_link_libraries(covoxslam_core PUBLIC CUDA::cudart)
    set_target_properties(covoxslam_core PROPERTIES CUDA_STANDARD 20
        CUDA_STANDARD_REQUIRED ON CUDA_EXTENSIONS OFF CUDA_RUNTIME_LIBRARY Shared)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/covoxslam_build_metadata.txt"
        CONTENT "cmake=${CMAKE_VERSION}\nhost=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\ncuda=${CMAKE_CUDA_COMPILER_VERSION}\narchitecture=${CMAKE_CUDA_ARCHITECTURES}\nhost_standard=23\ncuda_standard=20\n")
else()
    target_sources(covoxslam_core PRIVATE src/cuda_mapping_stub.cpp)
    target_sources(covoxslam_core PRIVATE src/sensor_grid_cuda_stub.cpp)
    target_sources(covoxslam_core PRIVATE src/sensor_grid_map_stub.cpp)
endif()
target_include_directories(covoxslam_core PRIVATE src/input cuda)
include(cmake/ceronoceres.cmake)
include(cmake/urd.cmake)
if(NOT covoxslam_enable_cuda)
    message(STATUS "Host utilities only, coVoxSLAM mapping requires CUDA and URD")
endif()
target_sources(covoxslam_core PRIVATE src/storage/draft.cpp)
target_sources(covoxslam_core PRIVATE src/store_pipeline.cpp)
target_sources(covoxslam_core PRIVATE src/resident_pipeline.cpp)
if(MSVC)
    set_source_files_properties(src/store_pipeline.cpp PROPERTIES COMPILE_OPTIONS /bigobj)
endif()
target_include_directories(covoxslam_core PUBLIC include)
if(MSVC)
    target_compile_options(covoxslam_core PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
endif()

add_executable(covoxslam apps/main.cpp)
 target_link_libraries(covoxslam PRIVATE covoxslam_core)
target_compile_definitions(covoxslam PRIVATE covoxslam_version="${PROJECT_VERSION}")
add_executable(covoxslam_cuda apps/cuda_main.cpp)
target_link_libraries(covoxslam_cuda PRIVATE covoxslam_core)
